function drawResult(result) { const width = video.clientWidth; const height = video.clientHeight; overlay.width = width; overlay.height = height; ctx.clearRect( 0, 0, width, height ); /* ===================================== 後鏡頭 永遠使用智慧風景構圖 不做人臉判斷 不會再跳成人像模式 ===================================== */ if ( currentFacingMode === "environment" ) { faceInfo.innerText = ""; modeInfo.innerText = ""; drawLandscapeComposition( width, height ); return; } /* ===================================== 前鏡頭 才做人臉偵測 ===================================== */ const scaleX = width / video.videoWidth; const scaleY = height / video.videoHeight; const validFaces = []; if (result.detections) { for ( const detection of result.detections ) { const box = detection.boundingBox; const x = box.originX * scaleX; const y = box.originY * scaleY; const w = box.width * scaleX; const h = box.height * scaleY; if ( w < width * 0.10 || h < height * 0.10 ) { continue; } validFaces.push({ x, y, w, h, centerX: x + w / 2, centerY: y + h / 2 }); } } /* 前鏡頭有人 */ if ( validFaces.length > 0 ) { drawPeopleComposition( validFaces, width, height ); } else { /* 前鏡頭暫時沒偵測到人 不顯示「找不到人臉」 */ faceInfo.innerText = ""; modeInfo.innerText = ""; guideInfo.innerText = ""; scoreInfo.innerText = ""; } }
